4/// Email templates.
5///
6/// If the `email-templates` feature flag is enabled, the methods of this
7/// struct return strings formatted for sending `multipart/HTML` emails. The
8/// feature is not enabled, the templates are serialized as JSON. The
9/// `email-templates` feature *should be enabled* in the production environment.
10///
11/// See also: [`TokenScope`].
12///
13/// [`TokenScope`]: `crate::token::TokenScope`
14#[allow(missing_docs)]
15#[derive(Clone, Debug, Serialize)]
16pub enum Template {
17    /// See [`CompleteSignup`].
18    ///
19    /// [`CompleteSignup`]: `crate::token::TokenScope::CompleteSignup`
20    ConfirmSignup {
21        username: String,
22        token: String,
23        application: Option<(ApplicationId, String)>,
24    },
25
26    /// See [`AddEmail`].
27    ///
28    /// [`AddEmail`]: `crate::token::TokenScope::AddEmail`
29    ConfirmNewEmail {
30        username: String,
31        email: String,
32        token: String,
33    },
34
35    /// See [`ChangePrimaryEmail`].
36    ///
37    /// [`ChangePrimaryEmail`]: `crate::token::TokenScope::ChangePrimaryEmail`
38    ConfirmPrimaryEmailChange {
39        username: String,
40        current_primary_email: String,
41        new_primary_email: String,
42        token: String,
43    },
44}
45
46#[cfg(feature = "email-templates")]
47macro_rules! get_template {
48    (subject $name:literal) => {
49        get_template!(@ "-subject.txt" $name)
50    };
51
52    (body_html $name:literal) => {
53        get_template!(@ ".html" $name)
54    };
55
56    (body_text $name:literal) => {
57        get_template!(@ ".txt" $name)
58    };
59
60    (@ $exten:literal $name:expr) => {
61        include_str!(concat!(env!("OUT_DIR"), "/email-templates_", $name, $exten))
62    };
63}
64
65#[cfg(feature = "email-templates")]
66macro_rules! build_email {
67    (subject $template:expr) => {
68        match $template {
69            Self::ConfirmSignup { .. } => get_template!(subject "confirm-signup"),
70            Self::ConfirmNewEmail { .. } => get_template!(subject "confirm-new-email"),
71            Self::ConfirmPrimaryEmailChange { .. } =>
72                get_template!(subject "confirm-primary-email-change"),
73        }
74    };
75    ($ty:tt $template:expr, $callback_url:expr) => {
76        {
77            let callback_url = $callback_url;
78
79            match $template {
80                Self::ConfirmSignup { username, token, application } => {
81                    let callback_parameters = if let Some(application) = application {
82                        let url_encoded_redirect_url = urlencoding::encode(&application.1);
83
84                        format!(
85                            "application_id={}&redirect_url={}&token={}",
86                            application.0,
87                            url_encoded_redirect_url,
88                            token
89                        )
90                    } else {
91                        format!("token={}", token)
92                    };
93
94                    format!(
95                        get_template!($ty "confirm-signup"),
96                        callback_url = callback_url,
97                        username = username,
98                        callback_parameters = callback_parameters,
99                    )
100                },
101                Self::ConfirmNewEmail {
102                    username,
103                    token,
104                    ..
105                } => format!(
106                    get_template!($ty "confirm-new-email"),
107                    callback_url = callback_url,
108                    username = username,
109                    callback_parameters = format!("token={}", token)
110                ),
111                Self::ConfirmPrimaryEmailChange {
112                    username,
113                    current_primary_email,
114                    new_primary_email,
115                    token,
116                } => format!(
117                    get_template!($ty "confirm-primary-email-change"),
118                    callback_url = callback_url,
119                    username = username,
120                    current_primary_email = current_primary_email,
121                    new_primary_email = new_primary_email,
122                    callback_parameters = format!("token={}", token)
123                ),
124            }
125        }
126    };
127}

129impl Template {
130    /// Get subject of the template.
131    pub fn subject(&self) -> String {
132        #[cfg(feature = "email-templates")]
133        {
134            build_email!(subject self).to_string()
135        }
136
137        #[cfg(not(feature = "email-templates"))]
138        {
139            serde_variant::to_variant_name(&self).unwrap().to_string()
140        }
141    }
142
143    /// Get email body of the template.
144    pub fn body_html(&self, callback_url: &str) -> String {
145        #[cfg(feature = "email-templates")]
146        {
147            build_email!(body_html self, callback_url)
148        }
149
150        #[cfg(not(feature = "email-templates"))]
151        {
152            let json = serde_json::json!({
153                "callback_url": callback_url,
154                "template": self
155            });
156
157            serde_json::to_string(&json).unwrap()
158        }
159    }
160
161    /// Get plaintext email body of the template.
162    pub fn body_text(&self, callback_url: &str) -> String {
163        #[cfg(feature = "email-templates")]
164        {
165            build_email!(body_text self, callback_url)
166        }
167
168        #[cfg(not(feature = "email-templates"))]
169        {
170            self.body_html(callback_url)
171        }
172    }
173}
