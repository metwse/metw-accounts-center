15    /// Sign up a new account.
16    #[tracing::instrument(
17        level = "debug",
18        skip_all,
19        fields(account_id = tracing::field::Empty, username = signup_dto.username)
20    )]
21    pub async fn create(&self, signup_dto: &dto::request::Signup) -> ServiceResult<AccountId> {
22        let (is_email_taken_res, is_username_taken_rs) = tokio::join!(
23            self.repo.is_email_taken(&signup_dto.email),
24            self.repo.is_username_taken(&signup_dto.username)
25        );
26
27        if is_email_taken_res? {
28            return Err(ServiceError::EmailTaken);
29        }
30
31        if is_username_taken_rs? {
32            return Err(ServiceError::UsernameTaken);
33        }
34
35        let password_hash = password::hash(&signup_dto.password.base64_hash).await;
36
37        let mut transaction = self.repo.begin_transaction().await?;
38
39        let account_id = AccountId::unique();
40        let account = entity::Account {
41            account_id,
42            client_password_kdf: entity::ClientPasswordKdf::Base64EncodedPbkdf2Sha256 {
43                salt: signup_dto.password.pbkdf2_salt.clone(),
44                iterations: signup_dto.password.pbkdf2_iterations,
45                length: signup_dto.password.pbkdf2_length,
46            },
47            server_password_hash_algorithm: entity::ServerPasswordHashAlgorithm::Argon2id,
48            password_hash: Some(password_hash),
49
50            master_key_kek_kdf: entity::ClientPasswordKdf::None,
51            master_key_encryption_algorithm: entity::KeyEncryptionAlgorithm::None,
52            encrypted_master_key: None,
53            master_key_id: None,
54        };
55
56        transaction.insert(&account).await?;
57
58        transaction.insert_default_flags(account_id).await?;
59
60        transaction
61            .insert_username(account_id, &signup_dto.username, true)
62            .await?;
63
64        transaction.commit().await?;
65
66        tracing::Span::current().record("account_id", account_id.to_string());
67
68        Ok(account_id)
69    }

71    /// For use with login.
72    async fn login_with_credentials(
73        &self,
74        login_credentails: &dto::repo::OwnedLoginCredentials,
75        client_password_hash: &str,
76    ) -> ServiceResult<dto::service::Login> {
77        let Some(password_hash) = &login_credentails.password_hash else {
78            return Err(ServiceError::InvalidCredentials);
79        };
80
81        if match login_credentails.server_password_hash_algorithm {
82            entity::ServerPasswordHashAlgorithm::None if client_password_hash == password_hash => {
83                true
84            }
85
86            entity::ServerPasswordHashAlgorithm::Argon2id
87                if password::check(client_password_hash, password_hash).await =>
88            {
89                true
90            }
91
92            _ => false,
93        } {
94            Ok(dto::service::Login {
95                account_id: login_credentails.account_id,
96                is_email_verified: login_credentails.is_email_verified,
97            })
98        } else {
99            Err(ServiceError::InvalidCredentials)
100        }
101    }

103    /// Log into the account
104    #[tracing::instrument(
105        level = "debug",
106        skip_all,
107        fields(account_identifier = ?login_dto.account_identifier)
108    )]
109    pub async fn verify_login(
110        &self,
111        login_dto: &dto::request::Login,
112    ) -> ServiceResult<dto::service::Login> {
113        let Some(login_credentails) = (match &login_dto.account_identifier {
114            dto::request::AccountIdentifier::Username(username) => {
115                self.repo
116                    .get_login_credentials_by_username(&username.username)
117                    .await?
118            }
119            dto::request::AccountIdentifier::Email(email) => {
120                self.repo
121                    .get_login_credentials_by_email(&email.email)
122                    .await?
123            }
124            dto::request::AccountIdentifier::AccountId(account_id) => {
125                self.repo
126                    .get_login_credentials_by_account_id(*account_id)
127                    .await?
128            }
129        }) else {
130            return Err(ServiceError::AccountNotFound);
131        };
132
133        self.login_with_credentials(&login_credentails, &login_dto.client_password_hash)
134            .await
135    }

171        let (username, username_aliases, email, secondary_emails) = tokio::try_join!(
172            self.repo.get_primary_username(account_id),
173            self.repo.get_nonexpiring_username_aliases(account_id),

175            self.repo.get_secondary_emails(account_id),
176        )?;
177
178        Ok(dto::response::Account {
179            account_id,
180
181            username,
182            email,
183
184            username_aliases,
185            secondary_emails,
186        })

189    /// Change account's password.
190    #[tracing::instrument(level = "debug", skip(self, change_password_dto))]
191    pub async fn change_password(
192        &self,
193        account_id: AccountId,
194        change_password_dto: &dto::request::ChangePassword,
195    ) -> ServiceResult<()> {
196        let Some(login_credentials) = self
197            .repo
198            .get_login_credentials_by_account_id(account_id)
199            .await?
200        else {
201            return Err(ServiceError::AccountNotFound);
202        };
203
204        self.login_with_credentials(
205            &login_credentials,
206            &change_password_dto.current_password_hash,
207        )
208        .await?;
209
210        let new_password = &change_password_dto.new_password;
211
212        let server_derived = password::hash(&new_password.base64_hash).await;
213
214        let mut transaction = self.repo.begin_transaction().await?;
215        transaction
216            .set_password_credentials(
217                account_id,
218                &server_derived,
219                &entity::ClientPasswordKdf::Base64EncodedPbkdf2Sha256 {
220                    salt: new_password.pbkdf2_salt.clone(),
221                    iterations: new_password.pbkdf2_iterations,
222                    length: new_password.pbkdf2_length,
223                },
224                &entity::ServerPasswordHashAlgorithm::Argon2id,
225            )
226            .await?;
227        transaction.commit().await?;
228
229        Ok(())
230    }
