20fn extract_bearer_token(req: &Request) -> Option<String> {
21    req.headers()
22        .get(header::AUTHORIZATION)
23        .and_then(|header| header.to_str().ok())
24        .and_then(|bearer_token| bearer_token.trim().strip_prefix("Bearer "))
25        .map(|token_str| token_str.to_string())
26}
27
28fn parse_basic_credentials(req: &Request) -> Option<(ApplicationId, String)> {
29    let token_str = req
30        .headers()
31        .get(header::AUTHORIZATION)
32        .and_then(|header| header.to_str().ok())
33        .and_then(|basic_token| basic_token.trim().strip_prefix("Basic "))
34        .and_then(|token_base64| BASE64_STANDARD.decode(token_base64).ok())
35        .and_then(|token_bytes| String::from_utf8(token_bytes).ok())?;
36
37    let (application_id_str, client_secret_str) = token_str.split_once(':')?;
38
39    Some((
40        application_id_str.parse().ok()?,
41        client_secret_str.to_string(),
42    ))
43}

45/// Authenticate a login session.
46#[tracing::instrument(level = "debug", skip_all)]
47pub async fn require_session(
48    State(state): State<AppState>,
49    mut req: Request,
50    next: Next,
51) -> AppMiddlewareResult<Response> {
52    let Some(token) = extract_bearer_token(&req) else {
53        return Err(HandlerError::Unauthorized)?;
54    };
55
56    match AuthenticationHandler(state)
57        .auth_session(token.to_string())
58        .await
59    {
60        Ok(account_id) => {
61            req.extensions_mut().insert(account_id);
62
63            Ok(next.run(req).await)
64        }
65        Err(_) => Err(HandlerError::Unauthorized)?,
66    }
67}

93/// Authenticate application.
94#[tracing::instrument(level = "debug", skip_all)]
95pub async fn require_application_credentials(
96    State(state): State<AppState>,
97    mut req: Request,
98    next: Next,
99) -> AppMiddlewareResult<Response> {
100    let Some((application_id, client_secret)) = parse_basic_credentials(&req) else {
101        return Err(HandlerError::Unauthorized)?;
102    };
103
104    ApplicationAccessHandler(state)
105        .authenticate(application_id, client_secret)
106        .await?;
107
108    req.extensions_mut().insert(application_id);
109
110    Ok(next.run(req).await)
111}

113/// Authenticate the login session before email verification.
114#[tracing::instrument(level = "debug", skip_all)]
115pub async fn require_application_ownership(
116    State(state): State<AppState>,
117    Extension(account_id): Extension<AccountId>,
118    Path(application_id): Path<ApplicationId>,
119    mut req: Request,
120    next: Next,
121) -> AppMiddlewareResult<Response> {
122    if ApplicationManagementHandler(state)
123        .authorize_ownership(application_id, account_id)
124        .await
125        .is_ok()
126    {
127        req.extensions_mut().insert(application_id);
128
129        Ok(next.run(req).await)
130    } else {
131        Err(HandlerError::Unauthorized)?
132    }
133}

135/// utoipa modifiers for middleware documentation.
136pub struct ApiDocSecurityAddon;
137
138impl Modify for ApiDocSecurityAddon {
139    fn modify(&self, openapi: &mut openapi::OpenApi) {
140        if let Some(components) = openapi.components.as_mut() {
141            components.add_security_scheme(
142                "session_jwt",
143                openapi::security::SecurityScheme::Http(
144                    openapi::security::HttpBuilder::new()
145                        .scheme(openapi::security::HttpAuthScheme::Bearer)
146                        .bearer_format("JWT")
147                        .build(),
148                ),
149            );
150
151            components.add_security_scheme(
152                "email_verification_session_jwt",
153                openapi::security::SecurityScheme::Http(
154                    openapi::security::HttpBuilder::new()
155                        .scheme(openapi::security::HttpAuthScheme::Bearer)
156                        .bearer_format("JWT")
157                        .build(),
158                ),
159            );
160        }
161    }
162}

164/// A key extractor that tries to get rate limiting key from the extension
165/// added by [`require_session`] or [`require_email_verification_session`].
166#[derive(Clone, Default)]
167pub struct GovernorAccountIdKeyExtractor;
168
169impl KeyExtractor for GovernorAccountIdKeyExtractor {
170    type Key = AccountId;
171
172    fn extract<T>(&self, req: &Request<T>) -> Result<Self::Key, tower_governor::GovernorError> {
173        Ok(*req.extensions().get::<AccountId>().unwrap())
