3use crate::{
4    middleware::access_control::{ApiDocSecurityAddon, require_application_credentials},
5    res::{AppJson, AppResult},

7use axum::{
8    Extension, Router,
9    extract::{Path, State},
10    middleware,
11    routing::{get, post},
12};
13use service::{
14    AppState, dto,
15    handlers::ApplicationAccessHandler,
16    id::{AccountId, ApplicationId},
17};
18use utoipa::OpenApi;
19
20#[utoipa::path(
21    post, path = "application/exchange",
22    request_body = dto::request::AuthorizationCode,
23    responses(
24        (status = OK, body = dto::response::AuthorizationCodeExchangeResult)
25    ),
26    security(("application_basic" = [])),
27)]
28async fn exchange_authorization_code(
29    State(state): State<AppState>,
30    Extension(application_id): Extension<ApplicationId>,
31    AppJson(authorization_code_dto): AppJson<dto::request::AuthorizationCode>,
32) -> AppResult<dto::response::AuthorizationCodeExchangeResult> {
33    Ok(AppJson(
34        ApplicationAccessHandler(state)
35            .exchange(application_id, authorization_code_dto)
36            .await?,
37    ))
38}
39
40#[utoipa::path(
41    get, path = "application/accounts/{account_id}",
42    params(("account_id" = AccountId, Path)),
43    responses(
44        (status = OK, body = dto::response::Account)
45    ),
46    security(("application_basic" = [])),
47)]
48async fn get_account(
49    State(state): State<AppState>,
50    Extension(application_id): Extension<ApplicationId>,
51    Path(account_id): Path<AccountId>,
52) -> AppResult<dto::response::Account> {
53    Ok(AppJson(
54        ApplicationAccessHandler(state)
55            .get_account(application_id, account_id)
56            .await?,
57    ))
58}
59
60pub fn routes(state: AppState) -> Router {
61    Router::new()
62        .route("/application/exchange", post(exchange_authorization_code))
63        .route("/application/accounts/{account_id}", get(get_account))
64        .route_layer(middleware::from_fn_with_state(
65            state.clone(),
66            require_application_credentials,
67        ))
68        .with_state(state)
69}
