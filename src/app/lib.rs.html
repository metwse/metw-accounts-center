1//! # metw-accounts-center app
2//!
3//! This crate contains the application binary, bootstrapping all handlers to
4//! the axum routers.
5//!
6//! *Note*: The Axum application is intended to run under a reverse proxy. The
7//! reverse proxy must provide the remote address via the `X-Real-IP` header.
8
9use axum::{Json, Router, routing::get};
10use serde::Serialize;
11use service::AppState;
12use std::{sync::LazyLock, time::Instant};
13use utoipa::{OpenApi, ToSchema};
14
15/// API routes.
16pub mod routes;
17
18/// API middleware.
19pub mod middleware;
20
21/// API results.
22pub mod res;
23
24/// OpenAPI documentation.
25#[derive(OpenApi)]
26#[openapi(
27    info(description = "metw-accounts-center API"),
28    servers((url = "http://localhost:3781", description = "Local server")),
29    external_docs(url = "https://metwse.github.io/metw-accounts-center/", description = "Crate documentation"),
30    paths(get_status),
31    nest(
32        (path = "/", api = routes::application_access::ApiDoc, tags = ["application_access"]),
33        (path = "/", api = routes::application_management::ApiDoc, tags = ["application_management"]),
34        (path = "/", api = routes::authentication::ApiDoc, tags = ["authentication"]),
35        (path = "/", api = routes::session::ApiDoc, tags = ["session"]),
36        (path = "/", api = routes::email_verification_session::ApiDoc, tags = ["email_verification_session"]),
37        (path = "/", api = routes::token_action::ApiDoc, tags = ["token_action"]),
38    )
39)]
40pub struct ApiDoc;
41
42/// Application status
43#[derive(Serialize, ToSchema)]
44struct AppStatus {
45    /// Status message.
46    pub message: String,
47    /// JavaScript that should be executed by clients if present.
48    pub patch: Option<String>,
49    /// Uptime of the API server, in seconds.
50    pub uptime: u64,
51}
52
53static STARTUP_TIME: LazyLock<Instant> = LazyLock::new(Instant::now);
54
55#[utoipa::path(
56    get, path = "/",
57    responses(
58        (status = OK, body = AppStatus)
59    )
60)]
61async fn get_status() -> Json<AppStatus> {
62    Json(AppStatus {
63        message: "OK".to_string(),
64        patch: None,
65        uptime: Instant::now().duration_since(*STARTUP_TIME).as_secs(),
66    })
67}
68
69/// Constructs the web API.
70pub fn app(state: AppState) -> Router {
71    let _ = *STARTUP_TIME;
72
73    Router::new()
74        .route("/", get(get_status))
75        .route("/openapi.json", get(async || Json(ApiDoc::openapi())))
76        .with_state(state.clone())
77        .merge(routes::application_access::routes(state.clone()))
78        .merge(routes::application_management::routes(state.clone()))
79        .merge(routes::authentication::routes(state.clone()))
80        .merge(routes::email_verification_session::routes(state.clone()))
81        .merge(routes::session::routes(state.clone()))
82        .merge(routes::token_action::routes(state.clone()))
83        .route_layer(axum::middleware::from_fn(
84            middleware::extract_real_ip::extract_real_ip,
85        ))
86        .fallback(async || res::AppError::NotFound)
87}
