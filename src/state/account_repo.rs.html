29    async fn get_login_credentials_by_email(
30        &self,
31        email: &str,
32    ) -> RepoResult<Option<dto::repo::OwnedLoginCredentials>> {
33        type ServerPasswordHashAlgorithmJson =
34            sqlx::types::Json<entity::ServerPasswordHashAlgorithm>;
35
36        let Some(login) = sqlx::query!(
37            r#"SELECT id, password_hash,
38                    server_password_hash_algorithm AS
39                        "server_password_hash_algorithm: ServerPasswordHashAlgorithmJson"
40                FROM accounts
41                WHERE id = (SELECT account_id FROM emails WHERE email = $1)"#,
42            email
43        )
44        .fetch_optional(&self.pool)
45        .await?
46        else {
47            return Ok(None);
48        };
49
50        let login = dto::repo::OwnedLoginCredentials {
51            id: login.id.into(),
52            is_email_verified: true,
53            password_hash: login.password_hash,
54            server_password_hash_algorithm: login.server_password_hash_algorithm.0,
55        };
56
57        Ok(Some(login))
58    }

139    async fn set_primary_email_if_current_is(
140        &self,
141        id: AccountId,
142        current_primary_email: &str,
143        new_primary_email: &str,
144    ) -> RepoResult<bool> {
145        let mut tx = self.pool.begin().await?;
146
147        let result1 = sqlx::query!(
148            "UPDATE emails SET is_primary = false
149                WHERE account_id = $1 AND is_primary = true AND email = $2 AND
150                      EXISTS(SELECT * FROM emails WHERE
151                             account_id = $1 AND is_primary = false AND email = $3)",
152            id as _,
153            current_primary_email,
154            new_primary_email
155        )
156        .execute(&mut *tx)
157        .await?;
158
159        if result1.rows_affected() == 0 {
160            return Ok(false);
161        }
162
163        let result2 = sqlx::query!(
164            "UPDATE emails SET is_primary = true
165                WHERE account_id = $1 AND is_primary = false AND email = $2",
166            id as _,
167            new_primary_email
168        )
169        .execute(&mut *tx)
170        .await?;
171
172        if result2.rows_affected() == 0 {
173            return Ok(false);
174        }
175
176        tx.commit().await?;
177
178        Ok(true)
179    }

252    async fn upsert_account(&mut self, id: AccountId, password_hash: &str) -> RepoResult<()> {
253        let client_password_kdf = Json(entity::ClientPasswordKdf::Pbkdf2Sha256 {
254            salt: "metw-accounts-center".to_string(),
255            iterations: 500000,
256            length: 256,
257        });
258        let server_password_hash_algorithm = Json(entity::ServerPasswordHashAlgorithm::Argon2id);
259
260        let master_key_kek_kdf = Json(entity::ClientPasswordKdf::None);
261        let master_key_encryption_algorithm = Json(entity::MasterKeyEncryptionAlgorithm::None);
262        let encrypted_master_key: Option<Vec<_>> = None;
263
264        sqlx::query!(
265            "INSERT INTO accounts (
266                    id,
267                    client_password_kdf, server_password_hash_algorithm, password_hash,
268                    master_key_kek_kdf, master_key_encryption_algorithm, encrypted_master_key
269                ) VALUES ($1, $2, $3, $4, $5, $6, $7)
270                ON CONFLICT (id)
271                DO UPDATE SET
272                    client_password_kdf = $2, server_password_hash_algorithm = $3, password_hash = $4,
273                    master_key_kek_kdf = $5, master_key_encryption_algorithm = $6, encrypted_master_key = $7
274                ",
275            id as _,
276            client_password_kdf as _,
277            server_password_hash_algorithm as _,
278            password_hash,
279            master_key_kek_kdf as _,
280            master_key_encryption_algorithm as _,
281            encrypted_master_key,
282        )
283        .execute(&mut *self.tx)
284        .await?;
285
286        Ok(())
287    }
